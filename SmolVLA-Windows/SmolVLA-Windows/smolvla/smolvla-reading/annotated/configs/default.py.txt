    # L0210 每个 EvalConfig 实例构造完成后运行参数校验和批次数处理。
    def __post_init__(self) -> None:
        # L0211 条件 self.recording_repo_id is not None and not self.recording 成立时进入下方分支。
        if self.recording_repo_id is not None and not self.recording:

    # L0220 根据可用的 CPU 核心和回合数计算自动批大小。
    def _auto_batch_size(self) -> int:
        # L0221 原文档中的说明或续行。
        """Pick batch_size based on CPU cores, capped by n_episodes."""
        # L0222 导入模块 math。
        import math
        # L0223 导入模块 os。
        import os

        # L0225 组合多个布尔判断，按 Python 的短路规则求值给 “cpu_cores” 赋值。
        cpu_cores = os.cpu_count() or 4
        # L0226 原作者对相邻代码的说明。
        # Each async env worker needs ~1 core; leave headroom for main process + inference.
        # L0227 调用 max，用它的结果给 “by_cpu” 赋值。
        by_cpu = max(1, math.floor(cpu_cores * 0.7))
        # L0228 调用 min，用它的结果作为当前调用的返回值。
        return min(by_cpu, self.n_episodes, 64)
