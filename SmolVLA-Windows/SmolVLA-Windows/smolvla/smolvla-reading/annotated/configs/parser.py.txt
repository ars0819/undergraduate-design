def _flatten_to_cli_args(d: dict, prefix: str = "") -> list[str]:
    # L0076 原文档中的说明或续行。
    """Recursively flatten a nested dict to CLI-style args (e.g. {"lr": 1e-4} -> ["--lr=0.0001"])."""
    # L0077 创建列表，按给定顺序保留元素给 “args”（位置参数序列） 赋值。
    args = []
    # L0078 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for key, value in d.items():
        # L0079 条件 key in (PATH_KEY, draccus.CHOICE_TYPE_KEY) 成立时进入下方分支。
        if key in (PATH_KEY, draccus.CHOICE_TYPE_KEY):
            # L0080 跳过这一轮剩余操作，进入下一轮循环。
            continue
        # L0081 用条件表达式在 if 前与 else 后的两个值之间选择给 “full_key” 赋值。
        full_key = f"{prefix}.{key}" if prefix else key
        # L0082 条件 isinstance(value, bool) 成立时进入下方分支。
        if isinstance(value, bool):
            # L0083 调用 lower，用它的结果给 “value” 赋值。
            value = str(value).lower()
        # L0084 条件 isinstance(value, dict) 成立时进入下方分支。

            # L0085 调用 args.extend，将可迭代对象展开后追加到末尾。

        elif isinstance(value, list):
            # L0087 调用 args.append，在列表末尾追加一个元素。
            args.append(f"--{full_key}={json.dumps(value)}")
        # L0088 前面的分支均未命中时，再判断 value is not None。
        elif value is not None:
            # L0089 调用 args.append，在列表末尾追加一个元素。
            args.append(f"--{full_key}={value}")
    # L0090 引用已有的值 args作为当前调用的返回值。
    return args

    # L0125 计算此行的右侧表达式给 “option” 赋值。
    option = f"--{arg_name}"
    # L0126 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for index, arg in enumerate(args):
        # L0127 条件 arg.startswith(f"{option}=") 成立时进入下方分支。
        if arg.startswith(f"{option}="):
            # L0128 调用 arg.removeprefix，用它的结果作为当前调用的返回值。
            return arg.removeprefix(f"{option}=")
        # L0129 条件 arg == option and index + 1 < len(args) and not args[index + 1].startswith("--") 成立时进入下方分支。
        if arg == option and index + 1 < len(args) and not args[index + 1].startswith("--"):
            # L0130 用方括号按键或索引读取 args作为当前调用的返回值。
            return args[index + 1]

    return None

                        # L0455 调用 filter_arg，用它的结果给 “cli_args” 赋值。
                        cli_args = filter_arg("config_path", cli_args)
                        # L0456 调用 argtype.from_pretrained，用它的结果给 “cfg”（配置对象） 赋值。
                        cfg = argtype.from_pretrained(config_path_cli, cli_args=cli_args)
                    # L0457 进入对应的另一条分支，注意用缩进找到配对的判断或异常块。
                    else:
                        # L0458 条件 config_path_cli 成立时进入下方分支。
                        if config_path_cli:
                            # L0459 调用 filter_arg，用它的结果给 “cli_args” 赋值。
                            cli_args = filter_arg("config_path", cli_args)
                        # L0460 用 draccus 根据数据类类型和终端参数生成 EvalPipelineConfig 实例。
                        cfg = draccus.parse(
                            # L0461 为“config_class”提供这一项的值，注意它所在的参数表或外层表达式。
                            config_class=argtype,
                            # L0462 为“config_path”提供这一项的值，注意它所在的参数表或外层表达式。
                            config_path=config_path_cli or config_path,
                            # L0463 为“args”（位置参数序列）提供这一项的值，注意它所在的参数表或外层表达式。
                            args=cli_args,
                        # L0464 闭合上方尚未结束的括号。
                        )
                # L0465 捕获这里列出的异常类型，然后执行下方的处理逻辑。
                except DecodingError as e:
                    # L0466 调用 print 执行操作。
                    print(f"error: {e}", file=sys.stderr)
                    # L0467 调用 sys.exit 执行操作。
                    sys.exit(1)
            # L0468 把解析得到的 cfg 传给真实函数，并保留其他显式传入的参数。
            response = fn(cfg, *args, **kwargs)
            # L0469 引用已有的值 response作为当前调用的返回值。
            return response

        # L0471 调用 cast，用它的结果作为当前调用的返回值。
        return cast(F, wrapper_inner)

    # L0473 调用 cast，用它的结果作为当前调用的返回值。
    return cast(Callable[[F], F], wrapper_outer)
