# L0040 把形状为 […,4] 的 xyzw 四元数转换成三分量轴角表示。
def _quat_to_axisangle(quat: np.ndarray) -> np.ndarray:
    # L0041 原文档中的说明或续行。
    """Convert xyzw quaternions to axis-angle like robosuite's `quat2axisangle`: zeros near the identity."""
    # L0042 先把输入转换成 NumPy float32 数组。
    quat = np.asarray(quat, dtype=np.float32)
    # L0043 抽取四元数标量部 w，并把浮点误差可能超出的值限制在 arccos 的定义域。
    w = np.clip(quat[..., 3], -1, 1)
    # L0044 调用 np.sqrt，用它的结果给 “den” 赋值。
    den = np.sqrt(np.maximum(1 - w * w, 0))
    # L0045 调用 np.divide，用它的结果给 “scale” 赋值。
    scale = np.divide(2 * np.arccos(w), den, out=np.zeros_like(den), where=den > 1e-10)
    # L0046 调用 astype，用它的结果作为当前调用的返回值。
    return (quat[..., :3] * scale[..., None]).astype(np.float32)


# L0049 LIBERO 的标准观测转换，构造 8 维状态并按约定翻转相机图。
def canonical_observation(obs: Mapping) -> dict[str, np.ndarray]:
    # L0050 原文档中的说明或续行。
    """Map a LIBERO observation to LeRobot keys, rotating images by 180 degrees and building the 8-D state."""
    # L0051 调用 dict，用它的结果给 “adapted” 赋值。
    adapted = dict(obs)
    # L0052 从复制的字典取出原始的嵌套机器人状态。
    state = adapted.pop("robot_state", None)
    # L0053 条件 state is not None 成立时进入下方分支。
    if state is not None:
        # L0054 拼接末端位置 3、姿态轴角 3、左右夹爪位置 2，共 8 维。
        adapted["agent_pos"] = np.concatenate(
            # L0055 延续上面的语句，(state["eef"]["pos"], _quat_to_axisangle(state["eef"]["quat"]), state["gripper"]["qpos"]),。
            (state["eef"]["pos"], _quat_to_axisangle(state["eef"]["quat"]), state["gripper"]["qpos"]),
            # L0056 为“axis”提供这一项的值，注意它所在的参数表或外层表达式。
            axis=-1,
        # L0057 延续上面的语句，).astype(np.float32)。
        ).astype(np.float32)
    # L0058 调用通用仿真观测整理器，这里明确启用图像旋转。
    return _canonical_observation(adapted, flip_images=True)


# L0061 定义函数 _get_suite。
def _get_suite(name: str) -> benchmark.Benchmark:
    # L0062 原文档中的说明或续行。
    """Instantiate a LIBERO suite by name with clear validation."""
    # L0063 调用 benchmark.get_benchmark_dict，用它的结果给 “bench” 赋值。
    bench = benchmark.get_benchmark_dict()
    # L0064 条件 name not in bench 成立时进入下方分支。
    if name not in bench:
        # L0065 抛出 ValueError，中断当前路径并交给上层的异常处理。
        raise ValueError(f"Unknown LIBERO suite '{name}'. Available: {', '.join(sorted(bench.keys()))}")
    # L0066 调用 此行左侧的可调用对象，用它的结果给 “suite” 赋值。
    suite = bench[name]()
    # L0067 条件 not getattr(suite, "tasks", None) 成立时进入下方分支。
    if not getattr(suite, "tasks", None):
        # L0068 抛出 ValueError，中断当前路径并交给上层的异常处理。
        raise ValueError(f"Suite '{name}' has no tasks.")
    # L0069 引用已有的值 suite作为当前调用的返回值。
    return suite


# L0072 定义函数 _select_task_ids。
def _select_task_ids(total_tasks: int, task_ids: Iterable[int] | None) -> list[int]:
    # L0073 原文档中的说明或续行。
    """Validate/normalize task ids. If None → all tasks."""
    # L0074 条件 task_ids is None 成立时进入下方分支。
    if task_ids is None:
        # L0075 调用 list，用它的结果作为当前调用的返回值。
        return list(range(total_tasks))
    # L0076 调用 sorted，用它的结果给 “ids” 赋值。
    ids = sorted({int(t) for t in task_ids})
    # L0077 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for t in ids:
        # L0078 条件 t < 0 or t >= total_tasks 成立时进入下方分支。
        if t < 0 or t >= total_tasks:
            # L0079 抛出 ValueError，中断当前路径并交给上层的异常处理。
            raise ValueError(f"task_id {t} out of range [0, {total_tasks - 1}].")
    # L0080 引用已有的值 ids作为当前调用的返回值。
    return ids

# L0090 定义函数 _load_initial_states。
def _load_initial_states(path: Path) -> np.ndarray:
    # L0091 原文档中的说明或续行。
    """Load a task's initial states, from the converted `.npy` file next to `path` when there is one."""
    # L0092 调用 path.with_suffix，用它的结果给 “portable” 赋值。
    portable = path.with_suffix(path.suffix + ".npy")
    # L0093 条件 portable.exists() 成立时进入下方分支。
    if portable.exists():
        # L0094 调用 np.load，用它的结果作为当前调用的返回值。
        return np.load(portable, allow_pickle=False)
    # L0095 条件 os.environ.get("LEROBOT_SIM_REQUIRE_NUMPY") == "1" 成立时进入下方分支。
    if os.environ.get("LEROBOT_SIM_REQUIRE_NUMPY") == "1":
        # L0096 抛出 FileNotFoundError，中断当前路径并交给上层的异常处理。
        raise FileNotFoundError(f"Convert initial-state asset before serving: {portable}")
    # L0097 原作者对相邻代码的说明。
    # Only the original torch-pickle assets need torch; sim Docker images ship converted .npy files.
    # L0098 导入模块 torch。
    import torch

    # L0100 调用 np.asarray，用它的结果作为当前调用的返回值。
    return np.asarray(torch.load(path, weights_only=False))  # nosec B614


# L0103 定义函数 get_task_init_states。
def get_task_init_states(task_suite: Any, i: int, is_libero_plus: bool = False) -> np.ndarray:
    # L0104 用方括号按键或索引读取 task_suite.tasks给 “task”（任务） 赋值。
    task = task_suite.tasks[i]
    # L0105 调用 Path，用它的结果给 “filename”（文件名） 赋值。
    filename = Path(task.init_states_file)
    # L0106 调用 Path，用它的结果给 “root” 赋值。
    root = Path(get_libero_path("init_states"))

    # L0108 条件 not is_libero_plus 成立时进入下方分支。
    if not is_libero_plus:
        # L0109 计算右侧表达式 root / task.problem_folder / filename.name给 “init_states_path” 赋值。
        init_states_path = root / task.problem_folder / filename.name
        # L0110 调用 _load_initial_states，用它的结果作为当前调用的返回值。
        return _load_initial_states(init_states_path)

    # L0112 原作者对相邻代码的说明。
    # LIBERO-plus: `_add_` / `_level` variants store extra-object layouts under
    # L0113 原作者对相邻代码的说明。
    # libero_newobj/ as a flat array that must be reshaped to (1, -1).
    # L0114 条件 "_add_" in filename.name or "_level" in filename.name 成立时进入下方分支。
    if "_add_" in filename.name or "_level" in filename.name:
        # L0115 计算右侧表达式 root / "libero_newobj" / task.problem_folder / filename.name给 “init_states_path” 赋值。
        init_states_path = root / "libero_newobj" / task.problem_folder / filename.name
        # L0116 调用 _load_initial_states，用它的结果给 “init_states” 赋值。
        init_states = _load_initial_states(init_states_path)
        # L0117 调用 init_states.reshape，用它的结果作为当前调用的返回值。
        return init_states.reshape(1, -1)

    # L0119 原作者对相邻代码的说明。
    # LIBERO-plus perturbation variants encode the perturbation in the filename
    # L0120 原作者对相邻代码的说明。
    # but on disk only the base `.pruned_init` exists — strip the suffix to match.
    # L0121 计算右侧表达式 _LIBERO_PERTURBATION_SUFFIX_RE.sub("", filename.stem) + filename.suffix给 “stripped” 赋值。
    stripped = _LIBERO_PERTURBATION_SUFFIX_RE.sub("", filename.stem) + filename.suffix
    # L0122 计算右侧表达式 root / task.problem_folder / stripped给 “init_states_path” 赋值。
    init_states_path = root / task.problem_folder / stripped
    # L0123 调用 _load_initial_states，用它的结果作为当前调用的返回值。
    return _load_initial_states(init_states_path)
