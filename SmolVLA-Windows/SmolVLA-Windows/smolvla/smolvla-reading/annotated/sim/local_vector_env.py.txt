# L0094 定义函数 _build_env。
def _build_env(sim_type: str, make: Callable[[], Any]) -> Any:
    # L0095 原文档中的说明或续行。
    """Build one env, or one vector env, with `make`; reject a vector env that autoresets."""
    # L0096 进入上下文管理器，缩进块结束时由它处理退出动作。
    with simulator_required(sim_type):
        # L0097 调用 make，用它的结果给 “env” 赋值。
        env = make()
    # L0098 条件 isinstance(env, gym.vector.VectorEnv) 成立时进入下方分支。
    if isinstance(env, gym.vector.VectorEnv):
        # L0099 调用 env.metadata.get，用它的结果给 “mode” 赋值。
        mode = env.metadata.get("autoreset_mode")
        # L0100 条件 mode is not _NO_AUTORESET 成立时进入下方分支。
        if mode is not _NO_AUTORESET:
            # L0101 调用 env.close，关闭对象或释放关联资源。
            env.close()
            # L0102 抛出 ValueError，中断当前路径并交给上层的异常处理。
            raise ValueError(
                # L0103 f 字符串片段，花括号中的表达式会被求值并插入文本。
                f"A vector env from vector_env_fns must not autoreset; set metadata['autoreset_mode'] to "
                # L0104 f 字符串片段，花括号中的表达式会被求值并插入文本。
                f"AutoresetMode.DISABLED (got {mode})"
            # L0105 闭合上方尚未结束的括号。
            )
    # L0106 引用已有的值 env作为当前调用的返回值。
    return env


# L0109 定义函数 _env_worker。
def _env_worker(pipe: Connection, make: CloudpickleWrapper) -> None:
    # L0110 原文档中的说明或续行。
    """Worker process: build one env, then run the commands sent on `pipe` until `close`."""
    # L0111 开始执行可能抛出异常的代码，后续 except 或 finally 提供处理。
    try:
        # L0112 调用 make，用它的结果给 “env” 赋值。
        env = make()
    # L0113 捕获这里列出的异常类型，然后执行下方的处理逻辑。
    except BaseException:
        # L0114 调用 pipe.send 执行操作。
        pipe.send((False, traceback.format_exc()))
        # L0115 退出当前调用并返回 None。
        return
    # L0116 调用 pipe.send 执行操作。
    pipe.send((True, None))
    # L0117 开始执行可能抛出异常的代码，后续 except 或 finally 提供处理。
    try:
        # L0118 持续执行下面的循环体，每轮开始都重新检查条件。
        while True:
            # L0119 调用 pipe.recv，用它的结果给 “(command, argument)” 赋值。
            command, argument = pipe.recv()
            # L0120 条件 command == "close" 成立时进入下方分支。
            if command == "close":
                # L0121 结束所在的最近一层循环。
                break
            # L0122 开始执行可能抛出异常的代码，后续 except 或 finally 提供处理。
            try:
                # L0123 条件 command == "reset" 成立时进入下方分支。
                if command == "reset":
                    # L0124 声明“result”的类型为 Any，调用 env.reset，用它的结果作为初始值。
                    result: Any = env.reset(seed=argument)
                # L0125 前面的分支均未命中时，再判断 command == "step"。
                elif command == "step":
                    # L0126 调用 env.step，用它的结果给 “result” 赋值。
                    result = env.step(argument)
                # L0127 前面的分支均未命中时，再判断 command == "spaces"。
                elif command == "spaces":
                    # L0128 调用 _spaces，用它的结果给 “result” 赋值。
                    result = _spaces(env)
                # L0129 进入对应的另一条分支，注意用缩进找到配对的判断或异常块。
                else:  # "attrs"
                    # L0130 用字典推导式从可迭代对象逐项取值并构造结果给 “result” 赋值。
                    result = {name: getattr(env, name) for name in _ENV_ATTRS if hasattr(env, name)}
                    # L0131 调用 _max_episode_steps，用它的结果给 “result['_max_episode_steps']” 赋值。
                    result["_max_episode_steps"] = _max_episode_steps(env)
                # L0132 调用 pipe.send 执行操作。
                pipe.send((True, result))
            # L0133 捕获这里列出的异常类型，然后执行下方的处理逻辑。
            except Exception:
                # L0134 调用 pipe.send 执行操作。
                pipe.send((False, traceback.format_exc()))
    # L0135 离开 try 时执行这里的清理，即使前面的路径发生异常。
    finally:
        # L0136 调用 env.close，关闭对象或释放关联资源。
        env.close()
        # L0137 调用 pipe.close，关闭对象或释放关联资源。
        pipe.close()


# L0140 定义类 _ProcessEnv。
class _ProcessEnv:
    # L0141 原文档中的说明或续行。
    """One environment in its own process. A batch steps in parallel: send all, then receive all.

    # L0143 原文档中的说明或续行。
    Not Gymnasium's `AsyncVectorEnv`, which also steps finished environments (Gymnasium leaves that
    # L0144 原文档中的说明或续行。
    undefined) and builds an extra one in this process to read the spaces.
    # L0145 文档字符串的边界。
    """

    # L0147 定义方法 __init__。
    def __init__(self, make: Callable[[], Any]):
        # L0148 用条件表达式在 if 前与 else 后的两个值之间选择给 “start_method” 赋值。
        start_method = "forkserver" if "forkserver" in multiprocessing.get_all_start_methods() else "spawn"
        # L0149 声明“context”的类型为 Any，调用 multiprocessing.get_context，用它的结果作为初始值。
        context: Any = multiprocessing.get_context(start_method)
        # L0150 调用 context.Pipe，用它的结果给 “(self._pipe, child)” 赋值。
        self._pipe, child = context.Pipe()
        # L0151 调用 context.Process，用它的结果给 “self._process” 赋值。
        self._process = context.Process(
            # L0152 为“target”提供这一项的值，注意它所在的参数表或外层表达式。
            target=_env_worker, args=(child, CloudpickleWrapper(make)), daemon=True
        # L0153 闭合上方尚未结束的括号。
        )
        # L0154 调用 self._process.start，启动线程或进程。
        self._process.start()
        # L0155 调用 child.close，关闭对象或释放关联资源。
        child.close()
        # L0156 使用布尔值 False给 “self._close_requested” 赋值。
        self._close_requested = False

    # L0158 定义方法 ready。
    def ready(self) -> None:
        # L0159 原文档中的说明或续行。
        """Wait until the environment is built, then read its spaces and attributes."""
        # L0160 调用 self._receive 执行操作。
        self._receive()
        # L0161 调用 self._call，用它的结果给 “(self.observation_space, self.action_space)” 赋值。
        self.observation_space, self.action_space = self._call("spaces")
        # L0162 调用 self._refresh 执行操作。
        self._refresh()

    # L0164 定义方法 _exit_error。
    def _exit_error(self) -> RuntimeError:
        # L0165 调用 self._process.join，等待对应线程或进程结束。
        self._process.join(1)
        # L0166 调用 RuntimeError，用它的结果作为当前调用的返回值。
        return RuntimeError(
            # L0167 f 字符串片段，花括号中的表达式会被求值并插入文本。
            f"Environment process exited with code {self._process.exitcode}. If its simulator cannot run in worker "
            # L0168 字符串字面量，供上方的调用、容器或错误消息使用。
            "processes on this platform, run it in the main process (use_async_envs=false, e.g. "
            # L0169 字符串字面量，供上方的调用、容器或错误消息使用。
            "--eval.use_async_envs=false)"
        # L0170 闭合上方尚未结束的括号。
        )

    # L0172 定义方法 _receive。
    def _receive(self) -> Any:
        # L0173 开始执行可能抛出异常的代码，后续 except 或 finally 提供处理。
        try:
            # L0174 调用 self._pipe.recv，用它的结果给 “(ok, value)” 赋值。
            ok, value = self._pipe.recv()
        # L0175 捕获这里列出的异常类型，然后执行下方的处理逻辑。
        except EOFError:
            # L0176 抛出 self._exit_error，中断当前路径并交给上层的异常处理。
            raise self._exit_error() from None
        # L0177 条件 not ok 成立时进入下方分支。
        if not ok:
            # L0178 抛出 RuntimeError，中断当前路径并交给上层的异常处理。
            raise RuntimeError(f"Environment process failed: {value}")
        # L0179 引用已有的值 value作为当前调用的返回值。
        return value

    # L0181 定义方法 _call。
    def _call(self, command: str, argument: Any = None) -> Any:
        # L0182 调用 self.send 执行操作。
        self.send(command, argument)
        # L0183 调用 self._receive，用它的结果作为当前调用的返回值。
        return self._receive()

    # L0185 定义方法 _refresh。
    def _refresh(self) -> None:
        # L0186 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
        for name, value in self._call("attrs").items():
            # L0187 调用 setattr 执行操作。
            setattr(self, name, value)

    # L0189 定义方法 send。
    def send(self, command: str, argument: Any = None) -> None:
        # L0190 开始执行可能抛出异常的代码，后续 except 或 finally 提供处理。
        try:
            # L0191 调用 self._pipe.send 执行操作。
            self._pipe.send((command, argument))
        # L0192 捕获这里列出的异常类型，然后执行下方的处理逻辑。
        except BrokenPipeError:
            # L0193 抛出 self._exit_error，中断当前路径并交给上层的异常处理。
            raise self._exit_error() from None

    # L0195 定义方法 receive。
    def receive(self, refresh: bool = False) -> Any:
        # L0196 调用 self._receive，用它的结果给 “result” 赋值。
        result = self._receive()
        # L0197 条件 refresh 成立时进入下方分支。
        if refresh:
            # L0198 调用 self._refresh 执行操作。
            self._refresh()
        # L0199 引用已有的值 result作为当前调用的返回值。
        return result

    # L0201 定义方法 reset。
    def reset(self, seed: int | list[int | None] | None = None) -> Any:
        # L0202 调用 self.send 执行操作。
        self.send("reset", seed)
        # L0203 调用 self.receive，用它的结果作为当前调用的返回值。
        return self.receive(refresh=True)

    # L0205 定义方法 step。
    def step(self, action: np.ndarray) -> Any:
        # L0206 调用 self.send 执行操作。
        self.send("step", action)
        # L0207 调用 self.receive，用它的结果作为当前调用的返回值。
        return self.receive()

    # L0209 定义方法 request_close。
    def request_close(self) -> None:
        # L0210 原文档中的说明或续行。
        """Ask the process to close its environment and exit, without waiting. Asks only once."""
        # L0211 条件 self._close_requested 成立时进入下方分支。
        if self._close_requested:
            # L0212 退出当前调用并返回 None。
            return
        # L0213 使用布尔值 True给 “self._close_requested” 赋值。
        self._close_requested = True
        # L0214 进入上下文管理器，缩进块结束时由它处理退出动作。
        with contextlib.suppress(OSError):
            # L0215 调用 self._pipe.send 执行操作。
            self._pipe.send(("close", None))

    # L0217 定义方法 close。
    def close(self) -> None:
        # L0218 原文档中的说明或续行。
        """Close the environment and its process, terminating the process if it still runs after 10 s."""
        # L0219 调用 self.request_close 执行操作。
        self.request_close()
        # L0220 调用 self._process.join，等待对应线程或进程结束。
        self._process.join(10)
        # L0221 条件 self._process.is_alive() 成立时进入下方分支。
        if self._process.is_alive():
            # L0222 调用 self._process.terminate 执行操作。
            self._process.terminate()
        # L0223 调用 self._pipe.close，关闭对象或释放关联资源。
        self._pipe.close()


# L0226 用统一的批量接口包装同一任务的一组仿真环境。
class LocalVectorEnv(gym.vector.VectorEnv):
    # L0227 原文档中的说明或续行。
    """`num_envs` envs of one simulator task, with no autoreset.

    # L0229 原文档中的说明或续行。
    A finished env is not stepped until `reset`: it keeps its last observation and gets zero reward. With
    # L0230 原文档中的说明或续行。
