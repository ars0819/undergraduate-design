# L0041 对 Q 或 K 的每对特征按 token 位置旋转，给注意力提供位置关系。
def apply_rope(x, positions, max_wavelength=10_000):
    # L0042 文档字符串的边界。
    """
    # L0043 原文档中的说明或续行。此段讲给 Q 和 K 注入由 token 位置决定的旋转。
    Applies RoPE positions [B, L] to x [B, L, H, D].
    # L0044 文档字符串的边界。
    """
    # L0045 计算右侧表达式 x.shape[-1] // 2给 “d_half” 赋值。
    d_half = x.shape[-1] // 2
    # L0046 读取属性 x.device给 “device”（运算设备） 赋值。
    device = x.device
    # L0047 读取属性 x.dtype给 “dtype”（元素类型） 赋值。
    dtype = x.dtype
    # L0048 调用 x.to，用它的结果给 “x” 赋值。
    x = x.to(torch.float32)

    # L0050 计算右侧表达式 (2.0 / x.shape[-1]) * torch.arange(d_half, dtype=torch.float32, device=device)给 “freq_exponents” 赋值。
    freq_exponents = (2.0 / x.shape[-1]) * torch.arange(d_half, dtype=torch.float32, device=device)
    # L0051 计算右侧表达式 max_wavelength**freq_exponents给 “timescale” 赋值。
    timescale = max_wavelength**freq_exponents
    # L0052 计算右侧表达式 positions[..., None].to(torch.float32) / timescale[None, None, :].to(torch.float32)给 “radians” 赋值。
    radians = positions[..., None].to(torch.float32) / timescale[None, None, :].to(torch.float32)

    # L0054 按逗号区分的各轴分别切片，保留写成冒号的整轴给 “radians” 赋值。
    radians = radians[..., None, :]

    # L0056 调用 torch.sin，用它的结果给 “sin” 赋值。
    sin = torch.sin(radians)  # .to(dtype=dtype)
    # L0057 调用 torch.cos，用它的结果给 “cos” 赋值。
    cos = torch.cos(radians)  # .to(dtype=dtype)

    # L0059 调用 x.split，用它的结果给 “(x1, x2)” 赋值。
    x1, x2 = x.split(d_half, dim=-1)
    # L0060 调用 torch.empty_like，用它的结果给 “res” 赋值。
    res = torch.empty_like(x)
    # L0061 计算右侧表达式 x1 * cos - x2 * sin给 “res[..., :d_half]” 赋值。
    res[..., :d_half] = x1 * cos - x2 * sin
    # L0062 计算右侧表达式 x2 * cos + x1 * sin给 “res[..., d_half:]” 赋值。
    res[..., d_half:] = x2 * cos + x1 * sin

    # L0064 调用 res.to，用它的结果作为当前调用的返回值。
    return res.to(dtype)


# L0067 定义函数 get_intermediate_size。
def get_intermediate_size(hidden_dim, ffn_dim_multiplier=4, multiple_of=256):
    # L0068 调用 int，用它的结果给 “hidden_dim” 赋值。
    hidden_dim = int(2 * hidden_dim / 3)
    # L0069 调用 int，用它的结果给 “hidden_dim” 赋值。
    hidden_dim = int(ffn_dim_multiplier * hidden_dim)
    # L0070 计算右侧表达式 multiple_of * ((hidden_dim + multiple_of - 1) // multiple_of)给 “hidden_dim” 赋值。
    hidden_dim = multiple_of * ((hidden_dim + multiple_of - 1) // multiple_of)
    # L0071 引用已有的值 hidden_dim作为当前调用的返回值。
    return hidden_dim

        freeze_vision_encoder: bool = False,

        attention_mode: str = "self_attn",
        # L0082 为“num_expert_layers”提供这一项的值，注意它所在的参数表或外层表达式。
        num_expert_layers: int = -1,
        # L0083 为“num_vlm_layers”提供这一项的值，注意它所在的参数表或外层表达式。
        num_vlm_layers: int = -1,
        # L0084 为“self_attn_every_n_layers”提供这一项的值，注意它所在的参数表或外层表达式。
        self_attn_every_n_layers: int = -1,
        # L0085 为“expert_width_multiplier”提供这一项的值，注意它所在的参数表或外层表达式。
        expert_width_multiplier: float = 0.5,
        # L0086 为“device”（运算设备）提供这一项的值，注意它所在的参数表或外层表达式。
        device: str = "auto",
    # L0087 闭合上方尚未结束的括号。
    ):
        # L0088 调用 __init__ 执行操作。
        super().__init__()
        # L0089 调用 require_package 执行操作。
        require_package("transformers", extra="smolvla")
        # L0090 该开关控制加载预训练视觉语言骨干，本次 checkpoint 设为 True。
        if load_vlm_weights:
            # L0091 调用 print 执行操作。
            print(f"Loading  {model_id} weights ...")
            # L0092 用 Transformers 创建骨干并读取已缓存的预训练参数。
            self.vlm = AutoModelForImageTextToText.from_pretrained(
                # L0093 “model_id”是当前括号中的一项，逗号将它与下一项分开。
                model_id,
                # L0094 构造这条路径时请求以 bfloat16 加载视觉语言骨干。运行时应逐个算子看 dtype。
                torch_dtype="bfloat16",
                # L0095 为“low_cpu_mem_usage”提供这一项的值，注意它所在的参数表或外层表达式。
                low_cpu_mem_usage=True,
            # L0096 闭合上方尚未结束的括号。
            )
            # L0097 读取属性 self.vlm.config给 “config”（配置对象） 赋值。
            config = self.vlm.config
        # L0098 进入对应的另一条分支，注意用缩进找到配对的判断或异常块。
        else:
            # L0099 调用 AutoConfig.from_pretrained，用它的结果给 “config”（配置对象） 赋值。
            config = AutoConfig.from_pretrained(model_id)
            # L0100 调用 SmolVLMForConditionalGeneration，用它的结果给 “self.vlm” 赋值。
            self.vlm = SmolVLMForConditionalGeneration(config=config)
        # L0101 加载与骨干配套的处理器，模型会读取它的 tokenizer 特殊 token ID。
        self.processor = AutoProcessor.from_pretrained(model_id)
        # L0102 正数表示保留前若干层。本次评测使用前 16 层。
        if num_vlm_layers > 0:
            # L0103 调用 print 执行操作。
            print(f"Reducing the number of VLM layers to {num_vlm_layers} ...")
            # L0104 对 text_model.layers 做切片，右端索引不包含在结果中给 “self.get_vlm_model().text_model.layers” 赋值。
            self.get_vlm_model().text_model.layers = self.get_vlm_model().text_model.layers[:num_vlm_layers]
        # L0105 保存实际使用的语言骨干层数。
        self.num_vlm_layers = len(self.get_vlm_model().text_model.layers)
        # L0106 引用已有的值 config给 “self.config” 赋值。
        self.config = config
        # L0107 原作者的注释。此处在讲加载视觉语言骨干并构造较窄的动作专家。
        # Smaller lm expert
        # L0108 深复制骨干文本配置。修改专家宽度时保留骨干原来的配置。
        lm_expert_config = copy.deepcopy(config.text_config)
        # L0109 读取属性 lm_expert_config.hidden_size给 “hidden_size” 赋值。
        hidden_size = lm_expert_config.hidden_size
        # L0110 用配置的比例缩小专家的隐藏宽度，再转为整数。

        # L0112 读取属性 self.num_vlm_layers给 “lm_expert_config.num_hidden_layers” 赋值。
        lm_expert_config.num_hidden_layers = self.num_vlm_layers
        # L0113 条件 num_expert_layers > 0 成立时进入下方分支。

                f"Number of layers in the VLM {len(self.get_vlm_model().text_model.layers)} are not multiple of num_expert_layers {num_expert_layers}"
            # L0116 闭合上方尚未结束的括号。
            )
            # L0117 显式设置专家层数时走这个分支。本次 0 让专家初始化继承前面设定的骨干层数。

        # L0118 根据调整后的配置创建动作专家。完整策略 checkpoint 稍后恢复相关权重。
        self.lm_expert = AutoModel.from_config(lm_expert_config)

        # L0120 调用 len，用它的结果给 “self.num_expert_layers” 赋值。
        self.num_expert_layers = len(self.lm_expert.layers)
        # L0121 引用已有的值 self_attn_every_n_layers给 “self.self_attn_every_n_layers” 赋值。
        self.self_attn_every_n_layers = self_attn_every_n_layers
        # L0122 交叉注意力模式需要让一部分专家层接受来自视觉语言骨干的 K/V 宽度。
        if "cross" in attention_mode:
            # L0123 原作者的注释。此处在讲加载视觉语言骨干并构造较窄的动作专家。
            # Reshape qkv projections to have the same input dimension as the vlm
            # L0124 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
            for layer_idx in range(len(self.lm_expert.layers)):
                # L0125 按层号取余。选作自注意力的层保留原 K/V 投影。
                if self.self_attn_every_n_layers > 0 and layer_idx % self.self_attn_every_n_layers == 0:
                    # L0126 跳过这一轮剩余操作，进入下一轮循环。
                    continue
                # L0127 给交叉注意力层重建 K 投影，让它接收骨干的 K 向量。
                self.lm_expert.layers[layer_idx].self_attn.k_proj = nn.Linear(
                    # L0128 延续上面的语句，config.text_config.num_key_value_heads * config.text_config.head_dim,。本段处理加载视觉语言骨干并构造较窄的动作专家。
                    config.text_config.num_key_value_heads * config.text_config.head_dim,
                    # L0129 延续上面的语句，lm_expert_config.num_key_value_heads * lm_expert_config.head_dim,。本段处理加载视觉语言骨干并构造较窄的动作专家。
                    lm_expert_config.num_key_value_heads * lm_expert_config.head_dim,
                    # L0130 为“bias”提供这一项的值，注意它所在的参数表或外层表达式。

                # L0131 闭合上方尚未结束的括号。
                )
                # L0132 给交叉注意力层重建 V 投影，让它接收骨干的 V 向量。
                self.lm_expert.layers[layer_idx].self_attn.v_proj = nn.Linear(
                    # L0133 延续上面的语句，config.text_config.num_key_value_heads * config.text_config.head_dim,。本段处理加载视觉语言骨干并构造较窄的动作专家。
                    config.text_config.num_key_value_heads * config.text_config.head_dim,
                    # L0134 延续上面的语句，lm_expert_config.num_key_value_heads * lm_expert_config.head_dim,。本段处理加载视觉语言骨干并构造较窄的动作专家。
                    lm_expert_config.num_key_value_heads * lm_expert_config.head_dim,
                    # L0135 为“bias”提供这一项的值，注意它所在的参数表或外层表达式。
                    bias=lm_expert_config.attention_bias,
                # L0136 闭合上方尚未结束的括号。
                )
        # L0137 原作者的注释。此处在讲加载视觉语言骨干并构造较窄的动作专家。
        # Remove unused embed_tokens

        # L0140 读取属性 self.config.text_config.num_attention_heads给 “self.num_attention_heads” 赋值。
        self.num_attention_heads = self.config.text_config.num_attention_heads
        # L0141 读取属性 self.config.text_config.num_key_value_heads给 “self.num_key_value_heads” 赋值。
        self.num_key_value_heads = self.config.text_config.num_key_value_heads

        # L0143 引用已有的值 freeze_vision_encoder给 “self.freeze_vision_encoder” 赋值。
        self.freeze_vision_encoder = freeze_vision_encoder

        self.attention_mode = attention_mode
        # L0146 读取属性 lm_expert_config.hidden_size给 “self.expert_hidden_size” 赋值。
        self.expert_hidden_size = lm_expert_config.hidden_size

    # L0152 定义方法 set_requires_grad。
    def set_requires_grad(self):
        # L0153 条件 self.freeze_vision_encoder 成立时进入下方分支。
        if self.freeze_vision_encoder:
            # L0154 调用 vision_model.eval，切换 PyTorch 模块到评测模式。

                params.requires_grad = False
        # L0157 条件 self.train_expert_only 成立时进入下方分支。
        if self.train_expert_only:
            # L0158 调用 self.vlm.eval，切换 PyTorch 模块到评测模式。
            self.vlm.eval()
            # L0159 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
            for params in self.vlm.parameters():
                # L0160 使用布尔值 False给 “params.requires_grad” 赋值。
                params.requires_grad = False
        # L0161 进入对应的另一条分支，注意用缩进找到配对的判断或异常块。
        else:
            # L0162 训练路径的说明。
            # To avoid unused params issue with distributed training
            # L0163 创建列表，按给定顺序保留元素给 “last_layers” 赋值。
            last_layers = [self.num_vlm_layers - 1]
            # L0164 条件 ( 成立时进入下方分支。
            if (
                # L0165 延续上面的语句，self.num_vlm_layers != self.num_expert_layers。
                self.num_vlm_layers != self.num_expert_layers
                # L0166 续写上一行的布尔表达式，按 and / or 的短路规则判断。
                and self.num_vlm_layers % self.num_expert_layers == 0
            # L0167 闭合上方尚未结束的括号。
            ):
                # L0168 调用 last_layers.append，在列表末尾追加一个元素。
                last_layers.append(self.num_vlm_layers - 2)
            # L0169 创建列表，按给定顺序保留元素给 “frozen_layers” 赋值。
            frozen_layers = [
                # L0170 字符串字面量，供上方的调用、容器或错误消息使用。
                "lm_head",
                # L0171 字符串字面量，供上方的调用、容器或错误消息使用。
                "text_model.norm.weight",
            # L0172 闭合上方尚未结束的括号。
            ]
            # L0173 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
            for layer in last_layers:
                # L0174 调用 frozen_layers.append，在列表末尾追加一个元素。
                frozen_layers.append(f"text_model.layers.{layer}.")

        # L0529 读取属性 self.num_key_value_heads给 “num_key_value_heads” 赋值。
        num_key_value_heads = self.num_key_value_heads
        # L0530 多个查询头可以共享一个 K/V 头。该比例决定需重复的组数。
        num_key_value_groups = num_att_heads // num_key_value_heads

        # L0532 用方括号按键或索引读取 key_states.shape给 “sequence_length” 赋值。
        sequence_length = key_states.shape[1]

        # L0534 调用 expand，用它的结果给 “key_states”（键向量 K） 赋值。
        key_states = key_states[:, :, :, None, :].expand(
            # L0535 延续上面的语句，batch_size, sequence_length, num_key_value_heads, num_key_value_groups, head_dim。本段处理用矩阵乘法、掩码和 softmax 计算注意力输出。
            batch_size, sequence_length, num_key_value_heads, num_key_value_groups, head_dim
        # L0536 闭合上方尚未结束的括号。
        )
        # L0537 调用 key_states.reshape，用它的结果给 “key_states”（键向量 K） 赋值。
        key_states = key_states.reshape(
            # L0538 延续上面的语句，batch_size, sequence_length, num_key_value_heads * num_key_value_groups, head_dim。本段处理用矩阵乘法、掩码和 softmax 计算注意力输出。
            batch_size, sequence_length, num_key_value_heads * num_key_value_groups, head_dim
        # L0539 闭合上方尚未结束的括号。
        )

        # L0541 调用 expand，用它的结果给 “value_states”（值向量 V） 赋值。
        value_states = value_states[:, :, :, None, :].expand(
            # L0542 延续上面的语句，batch_size, sequence_length, num_key_value_heads, num_key_value_groups, head_dim。本段处理用矩阵乘法、掩码和 softmax 计算注意力输出。
            batch_size, sequence_length, num_key_value_heads, num_key_value_groups, head_dim
        # L0543 闭合上方尚未结束的括号。
        )
        # L0544 调用 value_states.reshape，用它的结果给 “value_states”（值向量 V） 赋值。
        value_states = value_states.reshape(
            # L0545 延续上面的语句，batch_size, sequence_length, num_key_value_heads * num_key_value_groups, head_dim。本段处理用矩阵乘法、掩码和 softmax 计算注意力输出。
            batch_size, sequence_length, num_key_value_heads * num_key_value_groups, head_dim
        # L0546 闭合上方尚未结束的括号。
        )

        # L0548 原作者的注释。此处在讲用矩阵乘法、掩码和 softmax 计算注意力输出。
        # Attention here is upcasted to float32 to match the original eager implementation.
        # L0549 将 Q 提升为 float32 后计算点积，提高分数计算精度。
        query_states = query_states.to(dtype=torch.float32)
        # L0550 将 K 也转成 float32，与 Q 一起计算。
        key_states = key_states.to(dtype=torch.float32)

        # L0552 调用 query_states.transpose，用它的结果给 “query_states”（查询向量 Q） 赋值。
        query_states = query_states.transpose(1, 2)
        # L0553 调用 key_states.transpose，用它的结果给 “key_states”（键向量 K） 赋值。
        key_states = key_states.transpose(1, 2)

        # L0555 矩阵乘法计算每个查询与每个键的点积，得到 [B,头数,L查询,L键]。
        att_weights = torch.matmul(query_states, key_states.transpose(2, 3))
        # L0556 用头宽的平方根缩放点积，使分数尺度更稳定。
        att_weights *= head_dim**-0.5

        # L0558 调用 att_weights.to，用它的结果给 “att_weights” 赋值。
        att_weights = att_weights.to(dtype=torch.float32)
        # L0559 读取属性 min给 “big_neg” 赋值。
        big_neg = torch.finfo(att_weights.dtype).min  # -2.3819763e38  # See gemma/modules.py
        # L0560 把不可见键对应分数填为极小值。掩码在头轴广播。
        masked_att_weights = torch.where(attention_mask[:, None, :, :], att_weights, big_neg)
        # L0561 沿键位置轴用 softmax 归一化，每个查询得到一组对键的权重。
        probs = nn.functional.softmax(masked_att_weights, dim=-1)
        # L0562 将注意力概率转回 V 的 dtype 再做加权求和。
        probs = probs.to(dtype=value_states.dtype)

        # L0564 概率矩阵和 V 相乘，对可见键对应的值向量加权求和。
        att_output = torch.matmul(probs, value_states.permute(0, 2, 1, 3))

        # L0566 从 [B,头数,L查询,头宽] 变成 [B,L查询,头数,头宽]。
        att_output = att_output.permute(0, 2, 1, 3)
        # L0567 原作者的注释。此处在讲用矩阵乘法、掩码和 softmax 计算注意力输出。
        # we use -1 because sequence length can change
        # L0568 合并头数与头宽，得到 [B,L查询,隐藏维度]。
        att_output = att_output.reshape(batch_size, -1, num_key_value_heads * num_key_value_groups * head_dim)

        # L0570 引用已有的值 att_output作为当前调用的返回值。
        return att_output
