# L0104 定义函数 aloha_gripper_to_angular。
def aloha_gripper_to_angular(value):
    # L0105 原作者对相邻代码的说明。
    # Aloha transforms the gripper positions into a linear space. The following code
    # L0106 原作者对相邻代码的说明。
    # reverses this transformation to be consistent with smolvla which is pretrained in
    # L0107 原作者对相邻代码的说明。
    # angular space.
    # L0108 原作者对相邻代码的说明。
    #
    # L0109 原作者对相邻代码的说明。
    # These values are coming from the Aloha code:
    # L0110 原作者对相邻代码的说明。
    # PUPPET_GRIPPER_POSITION_OPEN, PUPPET_GRIPPER_POSITION_CLOSED
    # L0111 调用 unnormalize，用它的结果给 “value” 赋值。
    value = unnormalize(value, min_val=0.01844, max_val=0.05800)

    # L0113 原作者对相邻代码的说明。
    # This is the inverse of the angular to linear transformation inside the Interbotix code.
    # L0114 定义函数 linear_to_radian。
    def linear_to_radian(linear_position, arm_length, horn_radius):
        # L0115 计算右侧表达式 (horn_radius**2 + linear_position**2 - arm_length**2) / (2 * horn_radius * linear_position)给 “value” 赋值。
        value = (horn_radius**2 + linear_position**2 - arm_length**2) / (2 * horn_radius * linear_position)
        # L0116 调用 safe_arcsin，用它的结果作为当前调用的返回值。
        return safe_arcsin(value)

    # L0118 原作者对相邻代码的说明。

    # L0119 调用 linear_to_radian，用它的结果给 “value” 赋值。
    value = linear_to_radian(value, arm_length=0.036, horn_radius=0.022)

    # L0121 归一化或反归一化的说明，注意数据尺度的转换。
    # Normalize to [0, 1].
    # L0122 原作者对相邻代码的说明。

    # L0123 调用 normalize，用它的结果作为当前调用的返回值。
    return normalize(value, min_val=0.4, max_val=1.5)

# L0496 定义函数 pad_tensor。
def pad_tensor(tensor, max_len, pad_value=0):
    # L0497 文档字符串的边界。
    """
    # L0498 原文档中的说明或续行。
    Efficiently pads a tensor along sequence dimension to match max_len.

    # L0500 以下列出调用参数及其含义。
    Args:
        # L0501 原文档对“tensor”的说明。
        tensor (torch.Tensor): Shape (B, L, ...) or (B, L).
        # L0502 原文档对“max_len”的说明。
        max_len (int): Fixed sequence length.
        # L0503 原文档对“pad_value”的说明。
        pad_value (int/float): Value for padding.

    # L0505 以下说明调用者接收到的结果。
    Returns:
        # L0506 原文档对“torch.Tensor”的说明。
        torch.Tensor: Shape (B, max_len, ...) or (B, max_len).
    # L0507 文档字符串的边界。
    """
    # L0508 对 tensor.shape 做切片，右端索引不包含在结果中给 “(b, d)” 赋值。
    b, d = tensor.shape[:2]

    # L0510 原作者对相邻代码的说明。
    # Create a padded tensor of max_len and copy the existing values
    # L0511 调用 torch.full，用它的结果给 “padded_tensor” 赋值。
    padded_tensor = torch.full(
        # L0512 延续上面的语句，(b, max_len, *tensor.shape[2:]), pad_value, dtype=tensor.dtype, device=tensor.device。
        (b, max_len, *tensor.shape[2:]), pad_value, dtype=tensor.dtype, device=tensor.device
    # L0513 闭合上方尚未结束的括号。
    )
    # L0514 引用已有的值 tensor给 “padded_tensor[:, :d]” 赋值。
    padded_tensor[:, :d] = tensor  # Efficient in-place copy

    # L0516 引用已有的值 padded_tensor作为当前调用的返回值。
    return padded_tensor

        # L0745 条件 noise is None 成立时进入下方分支。
        if noise is None:
            # L0746 调用 self.sample_noise，用它的结果给 “noise”（采样起点噪声） 赋值。
            noise = self.sample_noise(actions.shape, actions.device)

        # L0748 条件 time is None 成立时进入下方分支。
        if time is None:
            # L0749 调用 self.sample_time，用它的结果给 “time” 赋值。
            time = self.sample_time(actions.shape[0], actions.device)

        # L0751 调用 make_flow_matching_inputs，用它的结果给 “(x_t, u_t, _)” 赋值。
        x_t, u_t, _ = make_flow_matching_inputs(actions, noise, time, convention=FlowConvention.NOISE_AT_ONE)
        # L0752 调用 self.embed_prefix，用它的结果给 “(prefix_embs, prefix_pad_masks, prefix_att_masks)” 赋值。
        prefix_embs, prefix_pad_masks, prefix_att_masks = self.embed_prefix(
            # L0753 延续上面的语句，images, img_masks, lang_tokens, lang_masks, state=state。本段处理构造训练输入并计算流匹配损失。
            images, img_masks, lang_tokens, lang_masks, state=state
        # L0754 闭合上方尚未结束的括号。
        )
        # L0755 调用 self.embed_suffix，用它的结果给 “(suffix_embs, suffix_pad_masks, suffix_att_masks)” 赋值。
        suffix_embs, suffix_pad_masks, suffix_att_masks = self.embed_suffix(x_t, time)

        # L0757 调用 torch.cat，用它的结果给 “pad_masks”（各 token 是否有效的掩码） 赋值。
        pad_masks = torch.cat([prefix_pad_masks, suffix_pad_masks], dim=1)
        # L0758 调用 torch.cat，用它的结果给 “att_masks”（注意力块边界标记） 赋值。
        att_masks = torch.cat([prefix_att_masks, suffix_att_masks], dim=1)

        # L0760 调用 make_att_2d_masks，用它的结果给 “att_2d_masks” 赋值。
        att_2d_masks = make_att_2d_masks(pad_masks, att_masks)
        # L0761 计算右侧表达式 torch.cumsum(pad_masks, dim=1) - 1给 “position_ids” 赋值。
        position_ids = torch.cumsum(pad_masks, dim=1) - 1
        # L0762 调用 self.vlm_with_expert.forward，用它的结果给 “((_, suffix_out), _)” 赋值。
        (_, suffix_out), _ = self.vlm_with_expert.forward(
            # L0763 为“attention_mask”提供这一项的值，注意它所在的参数表或外层表达式。
            attention_mask=att_2d_masks,
            # L0764 为“position_ids”提供这一项的值，注意它所在的参数表或外层表达式。
            position_ids=position_ids,
            # L0765 为“past_key_values”（各注意力层缓存的 K/V）提供这一项的值，注意它所在的参数表或外层表达式。
            past_key_values=None,
            # L0766 为“inputs_embeds”提供这一项的值，注意它所在的参数表或外层表达式。
            inputs_embeds=[prefix_embs, suffix_embs],
            # L0767 为“use_cache”提供这一项的值，注意它所在的参数表或外层表达式。
            use_cache=False,
        # L0768 闭合上方尚未结束的括号。
        )
        # L0769 按逗号区分的各轴分别切片，保留写成冒号的整轴给 “suffix_out” 赋值。
        suffix_out = suffix_out[:, -self.config.chunk_size :]
        # L0770 原作者的注释。此处在讲构造训练输入并计算流匹配损失。
        # Original openpi code, upcast attention output
        # L0771 调用 suffix_out.to，用它的结果给 “suffix_out” 赋值。
        suffix_out = suffix_out.to(dtype=torch.float32)
        # L0772 调用 self.action_out_proj，用它的结果给 “v_t”（当前速度场输出） 赋值。
        v_t = self.action_out_proj(suffix_out)
        # L0773 调用 F.mse_loss，用它的结果给 “losses” 赋值。
        losses = F.mse_loss(u_t, v_t, reduction="none")
        # L0774 引用已有的值 losses作为当前调用的返回值。
        return losses

    # L0776 推理采样入口，一次生成一块动作，在外层动作队列用空后才触发。
    def sample_actions(
        # L0777 self 接收当前实例，调用 obj.method() 时由 Python 自动传入。

        # L0782 “state”（机器人状态或当前步骤的状态）是当前括号中的一项，逗号将它与下一项分开。
        state,
        # L0783 为“noise”（采样起点噪声）提供这一项的值，注意它所在的参数表或外层表达式。
        noise=None,
        # L0784 星号展开可迭代对象或关键字字典。若在函数签名中则收集额外参数。
        **kwargs: Unpack[RTCActionSelectKwargs],
    # L0785 结束参数表。若箭头后有类型，它标注预期返回值，缩进块是函数体。

            actions_shape = (bsize, self.config.chunk_size, self.config.max_action_dim)

            noise = self.sample_noise(actions_shape, device)
        # L0793 进入对应的另一条分支，注意用缩进找到配对的判断或异常块。
        else:
            # L0794 调用者显式提供初始噪声时也将其移到同设备并转成 float32。

        # L0796 对当前观测做一次前缀嵌入，输出嵌入、有效位和注意力块标记。
        prefix_embs, prefix_pad_masks, prefix_att_masks = self.embed_prefix(
            # L0797 延续上面的语句，images, img_masks, lang_tokens, lang_masks, state=state。本段处理缓存观测条件并从初始噪声迭代生成动作块。
            images, img_masks, lang_tokens, lang_masks, state=state

        # L0800 对有效位沿序列轴累计计数，得到从零开始的有效 token 位置编号。
        prefix_position_ids = torch.cumsum(prefix_pad_masks, dim=1) - 1
        # L0801 原作者的注释。此处在讲缓存观测条件并从初始噪声迭代生成动作块。
        # Compute image and language key value cache
        # L0802 对观测前缀做一次预计算，取各层的键和值作为后续动作采样的缓存。

    # L0827 计算一个采样步骤的速度场，输入当前带噪动作块与观测前缀缓存。
    def denoise_step(
        # L0828 self 接收当前实例，调用 obj.method() 时由 Python 自动传入。
        self,
        # L0829 “prefix_pad_masks”是当前括号中的一项，逗号将它与下一项分开。
        prefix_pad_masks,
        # L0830 “past_key_values”（各注意力层缓存的 K/V）是当前括号中的一项，逗号将它与下一项分开。
        past_key_values,
        # L0831 “x_t”（采样时间 t 的当前动作块估计）是当前括号中的一项，逗号将它与下一项分开。
        x_t,
        # L0832 “timestep”是当前括号中的一项，逗号将它与下一项分开。
        timestep,
    # L0833 闭合上方尚未结束的括号。
    ):
        # L0834 原文档概述本段功能，根据条件和当前动作估计预测一次速度。
        """Apply one denoising step of the noise `x_t` at a given timestep."""
        # L0835 每个积分步骤都把当前动作估计与时间编码为动作后缀。
        suffix_embs, suffix_pad_masks, suffix_att_masks = self.embed_suffix(x_t, timestep)

        # L0837 用方括号按键或索引读取 suffix_pad_masks.shape给 “suffix_len” 赋值。
        suffix_len = suffix_pad_masks.shape[1]
        # L0838 用方括号按键或索引读取 prefix_pad_masks.shape给 “batch_size”（批大小） 赋值。
        batch_size = prefix_pad_masks.shape[0]
        # L0839 用方括号按键或索引读取 prefix_pad_masks.shape给 “prefix_len” 赋值。
        prefix_len = prefix_pad_masks.shape[1]
        # L0840 将观测 token 的有效位广播给所有动作查询，形状为 [B,L动作,L前缀]。
        prefix_pad_2d_masks = prefix_pad_masks[:, None, :].expand(batch_size, suffix_len, prefix_len)

        # L0842 调用 make_att_2d_masks，用它的结果给 “suffix_att_2d_masks” 赋值。
        suffix_att_2d_masks = make_att_2d_masks(suffix_pad_masks, suffix_att_masks)

        # L0844 沿键位置轴拼接前缀与动作的注意力掩码，后缀查询可按掩码查看两部分。
        full_att_2d_masks = torch.cat([prefix_pad_2d_masks, suffix_att_2d_masks], dim=2)
        # L0845 统计每个批次有效观测 token 的数目，作为后缀位置偏移。
        prefix_offsets = torch.sum(prefix_pad_masks, dim=-1)[:, None]
        # L0846 后缀位置在有效前缀位置后开始计数。
        position_ids = prefix_offsets + torch.cumsum(suffix_pad_masks, dim=1) - 1

        # L0848 用注意力模型处理当前动作后缀，并使用之前构造的观测前缀 K/V。
        outputs_embeds, _ = self.vlm_with_expert.forward(
            # L0849 为“attention_mask”提供这一项的值，注意它所在的参数表或外层表达式。
            attention_mask=full_att_2d_masks,
            # L0850 为“position_ids”提供这一项的值，注意它所在的参数表或外层表达式。
            position_ids=position_ids,
            # L0851 为“past_key_values”（各注意力层缓存的 K/V）提供这一项的值，注意它所在的参数表或外层表达式。
            past_key_values=past_key_values,
            # L0852 这次运行后缀路径，前缀输入位是 None。前缀条件来自缓存。
            inputs_embeds=[None, suffix_embs],
            # L0853 为“use_cache”提供这一项的值，注意它所在的参数表或外层表达式。
            use_cache=self.config.use_cache,
        # L0854 闭合上方尚未结束的括号。
        )
        # L0855 条件 past_key_values is not None 成立时进入下方分支。
        if past_key_values is not None:
            # L0856 原作者的注释。此处在讲根据条件和当前动作估计预测一次速度。
            # Self-attention layers append suffix K/V in place; restore the prefix for the next step.
            # L0857 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
            for layer in past_key_values.layers:
                # L0858 将曾被临时后缀扩展的层缓存裁回前缀长度，避免上一轮后缀混入下一轮。
                layer.crop(prefix_len - layer.get_seq_length())
        # L0859 取返回列表的第 2 项，即动作专家处理的后缀。
        suffix_out = outputs_embeds[1]
        # L0860 截取最后 chunk_size 个位置，确保对齐动作块时间轴。
        suffix_out = suffix_out[:, -self.config.chunk_size :]
        # L0861 将专家输出转为 float32，接下来转回动作空间。
        suffix_out = suffix_out.to(dtype=torch.float32)
        # L0862 对每个动作位置使用输出投影，速度预测形状为 [B,T,max_action_dim]。
        v_t = self.action_out_proj(suffix_out)
        # L0863 把速度场交回 Euler 循环，供它更新当前动作估计。
        return v_t
