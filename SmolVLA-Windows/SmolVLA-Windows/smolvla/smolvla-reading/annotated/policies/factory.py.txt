    # L0064 文档字符串的边界。
    """
    # L0065 原文档中的说明或续行。此段讲按配置注册名和实现模块的命名约定查找策略类。

    # L0071 原文档中的说明或续行。此段讲按配置注册名和实现模块的命名约定查找策略类。

    # L0099 把配置类的模块路径转换成同包的 modeling 模块路径并导入。

    policy_cls = getattr(module, cls_name, None)

def _import_sibling_policy_module(config_cls: type[PreTrainedConfig], prefix: str) -> ModuleType | None:
    # L0471 原文档中的说明或续行。
    """Import a config class' sibling ``{prefix}_*`` module, or None when that module does not exist."""
    # L0472 用配置类的 __module__ 拼出同目录的实现模块，动态导入。
    module_path = config_cls.__module__.replace("configuration_", f"{prefix}_")
    # L0473 开始执行可能抛出异常的代码，后续 except 或 finally 提供处理。
    try:
        # L0474 调用 importlib.import_module，用它的结果作为当前调用的返回值。
        return importlib.import_module(module_path)
    # L0475 捕获这里列出的异常类型，然后执行下方的处理逻辑。
    except ModuleNotFoundError as e:
        # L0476 条件 e.name == module_path 成立时进入下方分支。
        if e.name == module_path:
            # L0477 原作者对相邻代码的说明。
            # The sibling module itself does not exist for this policy type. A missing optional
            # L0478 原作者对相邻代码的说明。

            # L0479 原作者对相邻代码的说明。
            # actionable install hint stays visible.
            # L0480 设为空值 None作为当前调用的返回值。
            return None
        # L0481 抛出 异常，中断当前路径并交给上层的异常处理。
        raise
