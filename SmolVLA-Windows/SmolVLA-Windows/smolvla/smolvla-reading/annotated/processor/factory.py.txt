# L0187 定义函数 _reconnect_relative_absolute_steps。
def _reconnect_relative_absolute_steps(
    # L0188 延续上面的语句，preprocessor: PolicyProcessorPipeline, postprocessor: PolicyProcessorPipeline。
    preprocessor: PolicyProcessorPipeline, postprocessor: PolicyProcessorPipeline
# L0189 结束参数表。若箭头后有类型，它标注预期返回值，缩进块是函数体。
) -> None:
    # L0190 原文档中的说明或续行。
    """Wire AbsoluteActionsProcessorStep.relative_step to the RelativeActionsProcessorStep after deserialization.

    # L0192 原文档中的说明或续行。
    After a policy is loaded from disk, the preprocessor and postprocessor are reconstructed
    # L0193 原文档中的说明或续行。
    independently from their configs. AbsoluteActionsProcessorStep needs a live reference to
    # L0194 原文档中的说明或续行。
    the RelativeActionsProcessorStep so it can read the cached state at inference time.
    # L0195 原文档中的说明或续行。
    That reference is not serializable, so we re-establish it here after loading.
    # L0196 文档字符串的边界。
    """
    # L0197 检查前处理器中是否存在相对动作处理步骤，本次 checkpoint 没有这一步。
    relative_step = next((s for s in preprocessor.steps if isinstance(s, RelativeActionsProcessorStep)), None)
    # L0198 条件 relative_step is None 成立时进入下方分支。
    if relative_step is None:
        # L0199 退出当前调用并返回 None。
        return
    # L0200 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for step in postprocessor.steps:
        # L0201 条件 isinstance(step, AbsoluteActionsProcessorStep) and step.relative_step is None 成立时进入下方分支。
        if isinstance(step, AbsoluteActionsProcessorStep) and step.relative_step is None:
            # L0202 引用已有的值 relative_step给 “step.relative_step” 赋值。
            step.relative_step = relative_step


# L0205 从预训练模型目录或 Hub 缓存恢复策略的前后处理流水线。
def load_pretrained_policy_processors(
    # L0206 “pretrained_path”是当前括号中的一项，逗号将它与下一项分开。
    pretrained_path: str | Path,
    # L0207 后续参数必须按名字传入。
    *,
    # L0208 为“revision”（固定的 checkpoint 版本）提供这一项的值，注意它所在的参数表或外层表达式。
    revision: str | None = None,
    # L0209 为“preprocessor_overrides”提供这一项的值，注意它所在的参数表或外层表达式。
    preprocessor_overrides: dict[str, Any] | None = None,
    # L0210 为“postprocessor_overrides”提供这一项的值，注意它所在的参数表或外层表达式。
    postprocessor_overrides: dict[str, Any] | None = None,
    # L0211 为“preprocessor_config_filename”提供这一项的值，注意它所在的参数表或外层表达式。
    preprocessor_config_filename: str = f"{POLICY_PREPROCESSOR_DEFAULT_NAME}.json",
    # L0212 为“postprocessor_config_filename”提供这一项的值，注意它所在的参数表或外层表达式。
    postprocessor_config_filename: str = f"{POLICY_POSTPROCESSOR_DEFAULT_NAME}.json",
# L0213 延续上面的语句，) -> tuple[。
) -> tuple[
    # L0214 延续上面的语句，PolicyProcessorPipeline[dict[str, Any], dict[str, Any]],。
    PolicyProcessorPipeline[dict[str, Any], dict[str, Any]],
    # L0215 延续上面的语句，PolicyProcessorPipeline[PolicyAction, PolicyAction],。
    PolicyProcessorPipeline[PolicyAction, PolicyAction],
# L0216 闭合上方尚未结束的括号。
]:
    # L0217 原文档中的说明或续行。
    """Load a serialized policy pipeline pair and re-establish links that do not survive saving."""
    # L0218 读预训练前处理配置并创建有序的处理器对象。
    preprocessor = PolicyProcessorPipeline.from_pretrained(
        # L0219 为“pretrained_model_name_or_path”提供这一项的值，注意它所在的参数表或外层表达式。
        pretrained_model_name_or_path=pretrained_path,
        # L0220 为“config_filename”提供这一项的值，注意它所在的参数表或外层表达式。
        config_filename=preprocessor_config_filename,
        # L0221 用评测参数覆盖相机改名表和目标设备等保存字段。
        overrides=preprocessor_overrides,
        # L0222 把外部批量观测字典转换成处理器统一使用的 transition 结构。
        to_transition=batch_to_transition,
        # L0223 把前处理后的 transition 转回策略期望的批量字典。
        to_output=transition_to_batch,
        # L0224 为“revision”（固定的 checkpoint 版本）提供这一项的值，注意它所在的参数表或外层表达式。
        revision=revision,
    # L0225 闭合上方尚未结束的括号。
    )
    # L0226 读取对应 checkpoint 的后处理配置。
    postprocessor = PolicyProcessorPipeline.from_pretrained(
        # L0227 为“pretrained_model_name_or_path”提供这一项的值，注意它所在的参数表或外层表达式。
        pretrained_model_name_or_path=pretrained_path,
        # L0228 为“config_filename”提供这一项的值，注意它所在的参数表或外层表达式。
        config_filename=postprocessor_config_filename,
        # L0229 为“overrides”提供这一项的值，注意它所在的参数表或外层表达式。
        overrides=postprocessor_overrides,
        # L0230 把外部动作封装进 transition，以便复用同一套处理接口。
        to_transition=policy_action_to_transition,
        # L0231 执行完后从 transition 提取处理过的动作。
        to_output=transition_to_policy_action,
        # L0232 为“revision”（固定的 checkpoint 版本）提供这一项的值，注意它所在的参数表或外层表达式。
        revision=revision,
    # L0233 闭合上方尚未结束的括号。
    )
    # L0234 重新连接有需要的相对动作步骤。本次保存的步骤没有这对关联。
    _reconnect_relative_absolute_steps(preprocessor, postprocessor)
    # L0235 返回可直接调用的前处理和后处理对象。
    return preprocessor, postprocessor
