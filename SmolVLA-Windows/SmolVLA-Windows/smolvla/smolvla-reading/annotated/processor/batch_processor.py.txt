    - Single images (3D tensors).

    - Dictionaries of multiple images (3D tensors).
    # L0090 文档字符串的边界。

    # L0092 定义方法 observation。
    def observation(self, observation: dict[str, Tensor]) -> dict[str, Tensor]:
        # L0093 文档字符串的边界。
        """
        # L0094 原文档中的说明或续行。

        # L0101 文档字符串的边界。
        """
        # L0102 原作者对相邻代码的说明。
        # Process state observations - add batch dim if 1D
        # L0103 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
        for state_key in [OBS_STATE, OBS_ENV_STATE]:
            # L0104 条件 state_key in observation 成立时进入下方分支。
            if state_key in observation:
                # L0105 用方括号按键或索引读取 observation给 “state_value” 赋值。
                state_value = observation[state_key]
                # L0106 条件 isinstance(state_value, Tensor) and state_value.dim() == 1 成立时进入下方分支。
                if isinstance(state_value, Tensor) and state_value.dim() == 1:
                    # L0107 对一维状态向量加批次轴，从 [S] 变成 [1,S]。
                    observation[state_key] = state_value.unsqueeze(0)

        # L0109 原作者对相邻代码的说明。
        # Process single image observation - add batch dim if 3D
        # L0110 条件 OBS_IMAGE in observation 成立时进入下方分支。
        if OBS_IMAGE in observation:
            # L0111 用方括号按键或索引读取 observation给 “image_value” 赋值。
            image_value = observation[OBS_IMAGE]
            # L0112 条件 isinstance(image_value, Tensor) and image_value.dim() == 3 成立时进入下方分支。
            if isinstance(image_value, Tensor) and image_value.dim() == 3:
                # L0113 对一张 CHW 图像加批次轴，从 [C,H,W] 变成 [1,C,H,W]。
                observation[OBS_IMAGE] = image_value.unsqueeze(0)

        # L0115 原作者对相邻代码的说明。
        # Process multiple image observations - add batch dim if 3D
        # L0116 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
        for key, value in observation.items():
            # L0117 条件 key.startswith(f"{OBS_IMAGES}.") and isinstance(value, Tensor) and value.dim() == 3 成立时进入下方分支。
            if key.startswith(f"{OBS_IMAGES}.") and isinstance(value, Tensor) and value.dim() == 3:
                # L0118 调用 value.unsqueeze，用它的结果给 “observation[key]” 赋值。
                observation[key] = value.unsqueeze(0)
        # L0119 引用已有的值 observation作为当前调用的返回值。
        return observation

    # L0121 定义方法 transform_features。
    def transform_features(
        # L0122 延续上面的语句，self, features: dict[PipelineFeatureType, dict[str, PolicyFeature]]。
        self, features: dict[PipelineFeatureType, dict[str, PolicyFeature]]
    # L0123 结束参数表。若箭头后有类型，它标注预期返回值，缩进块是函数体。
    ) -> dict[PipelineFeatureType, dict[str, PolicyFeature]]:

        # L0135 引用已有的值 features作为当前调用的返回值。
        return features
