# L0115 定义函数 _env_features_to_dataset_features。
def _env_features_to_dataset_features(env_features: dict[str, PolicyFeature]) -> dict[str, dict[str, Any]]:
    # L0116 原文档中的说明或续行。
    """Convert EnvConfig.features to the dict format expected by LeRobotDataset.create()."""
    # L0117 声明“features”的类型为 dict[str, dict[str, Any]]，创建字典，用键查找对应的值作为初始值。
    features: dict[str, dict[str, Any]] = {}
    # L0118 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for key, ft in env_features.items():
        # L0119 调用 tuple，用它的结果给 “shape”（各轴的长度） 赋值。
        shape = tuple(ft.shape)
        # L0120 条件 ft.type is FeatureType.VISUAL 成立时进入下方分支。
        if ft.type is FeatureType.VISUAL:
            # L0121 创建字典，用键查找对应的值给 “features[key]” 赋值。
            features[key] = {"dtype": "video", "shape": shape, "names": ["height", "width", "channel"]}
        # L0122 进入对应的另一条分支，注意用缩进找到配对的判断或异常块。
        else:
            # L0123 创建字典，用键查找对应的值给 “features[key]” 赋值。
            features[key] = {"dtype": "float32", "shape": shape, "names": None}
    # L0124 创建字典，用键查找对应的值给 “features['next.reward']” 赋值。
    features["next.reward"] = {"dtype": "float32", "shape": (1,), "names": None}
    # L0125 创建字典，用键查找对应的值给 “features['next.success']” 赋值。
    features["next.success"] = {"dtype": "bool", "shape": (1,), "names": None}
    # L0126 创建字典，用键查找对应的值给 “features['next.done']” 赋值。
    features["next.done"] = {"dtype": "bool", "shape": (1,), "names": None}
    # L0127 引用已有的值 features作为当前调用的返回值。
    return features


# L0130 定义函数 _build_raw_frame。
def _build_raw_frame(
    # L0131 “raw_obs”是当前括号中的一项，逗号将它与下一项分开。
    raw_obs: dict,
    # L0132 “env_idx”是当前括号中的一项，逗号将它与下一项分开。
    env_idx: int,
    # L0133 “action”（单步动作）是当前括号中的一项，逗号将它与下一项分开。
    action: np.ndarray,
    # L0134 “reward”（单步奖励）是当前括号中的一项，逗号将它与下一项分开。
    reward: float,
    # L0135 “success”是当前括号中的一项，逗号将它与下一项分开。
    success: bool,
    # L0136 “done”是当前括号中的一项，逗号将它与下一项分开。
    done: bool,
    # L0137 “task”（任务）是当前括号中的一项，逗号将它与下一项分开。
    task: str,
    # L0138 “env_features”是当前括号中的一项，逗号将它与下一项分开。
    env_features: dict,
# L0139 结束参数表。若箭头后有类型，它标注预期返回值，缩进块是函数体。
) -> dict:
    # L0140 原文档中的说明或续行。
    """Build a dataset frame from raw env observations for one env index.

    # L0142 原文档中的说明或续行。
    Keys in the frame match the keys in env_features so they align with the
    # L0143 原文档中的说明或续行。
    dataset schema created by _env_features_to_dataset_features().
    # L0144 文档字符串的边界。
    """
    # L0145 声明“frame”的类型为 dict[str, Any]，创建字典，用键查找对应的值作为初始值。
    frame: dict[str, Any] = {}
    # L0146 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for key in env_features:
        # L0147 条件 key == ACTION 成立时进入下方分支。
        if key == ACTION:
            # L0148 跳过这一轮剩余操作，进入下一轮循环。
            continue
        # L0149 条件 key.startswith("next.") 成立时进入下方分支。
        if key.startswith("next."):
            # L0150 跳过这一轮剩余操作，进入下一轮循环。
            continue
        # L0151 条件 "pixels" in raw_obs and isinstance(raw_obs["pixels"], dict) 成立时进入下方分支。
        if "pixels" in raw_obs and isinstance(raw_obs["pixels"], dict):
            # L0152 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
            for cam_name, img in raw_obs["pixels"].items():
                # L0153 计算此行的右侧表达式给 “candidate” 赋值。
                candidate = f"{OBS_IMAGES}.{cam_name}"
                # L0154 条件 candidate == key 成立时进入下方分支。
                if candidate == key:
                    # L0155 用方括号按键或索引读取 img给 “frame[key]” 赋值。
                    frame[key] = img[env_idx]
            # L0156 条件 key in frame 成立时进入下方分支。
            if key in frame:
                # L0157 跳过这一轮剩余操作，进入下一轮循环。
                continue
        # L0158 条件 "pixels" in raw_obs and not isinstance(raw_obs["pixels"], dict) and key in ("pixels", OBS_IMAGE) 成立时进入下方分支。
        if "pixels" in raw_obs and not isinstance(raw_obs["pixels"], dict) and key in ("pixels", OBS_IMAGE):
            # L0159 用方括号按键或索引读取 容器给 “frame[key]” 赋值。
            frame[key] = raw_obs["pixels"][env_idx]
            # L0160 跳过这一轮剩余操作，进入下一轮循环。
            continue
        # L0161 条件 key in raw_obs and isinstance(raw_obs[key], np.ndarray) 成立时进入下方分支。
        if key in raw_obs and isinstance(raw_obs[key], np.ndarray):
            # L0162 用方括号按键或索引读取 容器给 “val” 赋值。
            val = raw_obs[key][env_idx]
            # L0163 条件 val.dtype == np.float64 成立时进入下方分支。
            if val.dtype == np.float64:
                # L0164 调用 val.astype，用它的结果给 “val” 赋值。
                val = val.astype(np.float32)
            # L0165 引用已有的值 val给 “frame[key]” 赋值。
            frame[key] = val
    # L0166 引用已有的值 action给 “frame[ACTION]” 赋值。
    frame[ACTION] = action
    # L0167 调用 np.atleast_1d，用它的结果给 “frame['next.reward']” 赋值。
    frame["next.reward"] = np.atleast_1d(np.float32(reward))
    # L0168 调用 np.atleast_1d，用它的结果给 “frame['next.success']” 赋值。
    frame["next.success"] = np.atleast_1d(np.bool_(success))
    # L0169 调用 np.atleast_1d，用它的结果给 “frame['next.done']” 赋值。
    frame["next.done"] = np.atleast_1d(np.bool_(done))
    # L0170 引用已有的值 task给 “frame['task']” 赋值。
    frame["task"] = task
    # L0171 引用已有的值 frame作为当前调用的返回值。
    return frame
