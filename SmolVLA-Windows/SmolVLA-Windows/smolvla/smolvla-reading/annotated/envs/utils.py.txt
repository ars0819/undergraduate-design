# L0040 定义函数 _convert_nested_dict。
def _convert_nested_dict(d):
    # L0041 创建字典，用键查找对应的值给 “result” 赋值。
    result = {}
    # L0042 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for k, v in d.items():
        # L0043 条件 isinstance(v, dict) 成立时进入下方分支。
        if isinstance(v, dict):
            # L0044 调用 _convert_nested_dict，用它的结果给 “result[k]” 赋值。
            result[k] = _convert_nested_dict(v)
        # L0045 前面的分支均未命中时，再判断 isinstance(v, np.ndarray)。
        elif isinstance(v, np.ndarray):
            # L0046 调用 torch.from_numpy，用它的结果给 “result[k]” 赋值。
            result[k] = torch.from_numpy(v)
        # L0047 进入对应的另一条分支，注意用缩进找到配对的判断或异常块。
        else:
            # L0048 引用已有的值 v给 “result[k]” 赋值。
            result[k] = v
    # L0049 引用已有的值 result作为当前调用的返回值。
    return result

            if not (c < h and c < w):

            # L0079 条件 img_tensor.dtype != torch.uint8 成立时进入下方分支。
            if img_tensor.dtype != torch.uint8:

        # L0094 引用已有的值 env_state给 “return_observations[OBS_ENV_STATE]” 赋值。
        return_observations[OBS_ENV_STATE] = env_state

    # L0096 条件 "agent_pos" in observations 成立时进入下方分支。

            # L0099 调用 agent_pos.unsqueeze，用它的结果给 “agent_pos” 赋值。

    # L0102 条件 "robot_state" in observations 成立时进入下方分支。
    if "robot_state" in observations:
        # L0103 调用 _convert_nested_dict，用它的结果给 “return_observations[f'{OBS_STR}.robot_state']” 赋值。
        return_observations[f"{OBS_STR}.robot_state"] = _convert_nested_dict(observations["robot_state"])

    # L0105 原作者对相邻代码的说明。
    # Handle IsaacLab Arena format: observations have 'policy' and 'camera_obs' keys
    # L0106 条件 "policy" in observations 成立时进入下方分支。
    if "policy" in observations:
        # L0107 用方括号按键或索引读取 observations给 “return_observations[f'{OBS_STR}.policy']” 赋值。
        return_observations[f"{OBS_STR}.policy"] = observations["policy"]

    # L0109 条件 "camera_obs" in observations 成立时进入下方分支。
    if "camera_obs" in observations:
        # L0110 用方括号按键或索引读取 observations给 “return_observations[f'{OBS_STR}.camera_obs']” 赋值。
        return_observations[f"{OBS_STR}.camera_obs"] = observations["camera_obs"]

    # L0112 原作者对相邻代码的说明。
    # Pass through any remaining ndarray/tensor keys not already handled above,
    # L0113 原作者对相邻代码的说明。
    # so env plugins can expose extra observation keys via get_env_processors().
    # L0114 计算此行的右侧表达式给 “_handled” 赋值。
    _handled = {"pixels", "environment_state", "agent_pos", "robot_state", "policy", "camera_obs"}
    # L0115 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。

            continue

# L0135 仿真已整理过键名时，把回合提供的批量 NumPy 观测转换为 Tensor。
def prepare_canonical_observation(observations: dict[str, np.ndarray]) -> dict[str, Tensor]:
    # L0136 原文档概述本段功能，将标准仿真 NumPy 观测转换为 torch Tensor。
    """Convert batched sim observations to tensors; uint8 BHWC images become float BCHW in [0, 1]."""
    # L0137 创建字典，用键查找对应的值给 “result” 赋值。
    result = {}
    # L0138 逐项迭代右侧的可迭代对象，并把每项绑定给 in 左侧的名字。
    for name, value in observations.items():
        # L0139 用 from_numpy 把 NumPy 数组封装为 CPU Tensor。数据类型沿用 NumPy。
        tensor = torch.from_numpy(value)
        # L0140 四轴 uint8 对应批量的彩色相机输入，源轴顺序为 [B,H,W,C]。
        if value.dtype == np.uint8 and value.ndim == 4:
            # L0141 把通道移到第二轴，再转为 float32 和 [0,1] 范围，设备仍为 CPU。
            tensor = tensor.permute(0, 3, 1, 2).contiguous().float() / 255
        # L0142 前面的分支均未命中时，再判断 tensor.is_floating_point()。
        elif tensor.is_floating_point():
            # L0143 保持非图像浮点特征原有的轴，将元素类型统一为 float32。
            tensor = tensor.float()
        # L0144 引用已有的值 tensor给 “result[name]” 赋值。
        result[name] = tensor
    # L0145 引用已有的值 result作为当前调用的返回值。
    return result
